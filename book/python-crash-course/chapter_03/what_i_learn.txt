fruit_list = [ 'apple' , 'banana' , 'carrot'] --> To create list 
fruit_list[0] = 'almond'                      --> change element at index 0 to almond
fruit_list.append('durian')                   --> add to next of last element
fruit_list.insert(0,'avocado')                --> add new element to index 0 with value 'avocado' and shift next element to left side 
del fruit_list[0]                             --> remove element at index 0
del fruit_list                                --> delete whole list 
last_element = fruit_list.pop()               --> to remove last element and remember it on last_element variable
first_element = fruit_list.pop(0)             --> to remove first element and remember it on first_element variable
fruit_list.remove('apple')                    --> the .remove() function is used to delete an item from a collection
fruit_list.sort()                             --> the .sort() function is used to sort alphabetical order (for strings) or numerical order (for numbers) permanently.
fruit_list.sort(reverse =  True)              --> sort order but reverse it
print(sorted(fruit_list))                     --> this is just a temporary sort in python.
fruit_list.reverse()                          --> to reverse the order of list
fruit_list.reverse()                          --> to revert the reverse
